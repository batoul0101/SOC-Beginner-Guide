def اكتشاف_هجمة_ssh(ملف_اللوج, حد=10):
    """
    اكتشاف هجمات SSH
    """
    من collections import Counter
    
    محاولات = Counter()
    
    with open(ملف_اللوج, 'r') as ملف:
        for سطر in ملف:
            if 'كلمة مرور فاشلة' in سطر:
                # استخراج IP
                import re
                تطابق = re.search(r'من (\d+\.\d+\.\d+\.\d+)', سطر)
                if تطابق:
                    ip = تطابق.group(1)
                    محاولات[ip] += 1
    
    # نتائج
    نتائج = []
    for ip, عدد in محاولات.items():
        if عدد >= حد:
            نتائج.append({
                'ip': ip,
                'محاولات': عدد,
                'حالة': 'مشبوه'
            })
    
    return نتائج
